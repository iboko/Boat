cmake_minimum_required(VERSION 3.20)
project(Boat VERSION 0.15.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BOAT_WITH_MAVLINK "Compile optional MAVLink 2 odometry encoder" OFF)
set(BOAT_MAVLINK_INCLUDE_DIR "" CACHE PATH "Directory containing common/mavlink.h")

add_library(boat_core STATIC src/navigation.cpp src/ardupilot.cpp src/ins.cpp src/sitl_link.cpp src/sensor_fixture.cpp src/shore_ranges.cpp src/shore_association.cpp src/marine_kinematics.cpp src/marine_dynamics.cpp)
target_include_directories(boat_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(boat_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(boat_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(BOAT_WITH_MAVLINK)
  if(NOT EXISTS "${BOAT_MAVLINK_INCLUDE_DIR}/common/mavlink.h")
    message(FATAL_ERROR "BOAT_MAVLINK_INCLUDE_DIR must contain common/mavlink.h")
  endif()
  target_compile_definitions(boat_core PUBLIC BOAT_WITH_MAVLINK=1)
  target_include_directories(boat_core SYSTEM PRIVATE "${BOAT_MAVLINK_INCLUDE_DIR}")
endif()

add_executable(boat_ins_demo apps/ins_demo.cpp)
target_link_libraries(boat_ins_demo PRIVATE boat_core)

add_executable(boat_marine_validation apps/marine_validation.cpp)
target_link_libraries(boat_marine_validation PRIVATE boat_core)

add_executable(boat_sim apps/sim_main.cpp)
target_link_libraries(boat_sim PRIVATE boat_core)
enable_testing()
add_executable(boat_tests tests/test_navigation.cpp)
target_link_libraries(boat_tests PRIVATE boat_core)
add_test(NAME boat_tests COMMAND boat_tests)

add_executable(boat_ins_tests tests/test_ins.cpp)
target_link_libraries(boat_ins_tests PRIVATE boat_core)
add_test(NAME boat_ins_tests COMMAND boat_ins_tests)
add_executable(boat_attitude_tests tests/test_attitude_aiding.cpp)
target_link_libraries(boat_attitude_tests PRIVATE boat_core)
add_test(NAME boat_attitude_tests COMMAND boat_attitude_tests)
add_executable(boat_wave_gyro_tests tests/test_waves_sensor_fusion.cpp)
target_link_libraries(boat_wave_gyro_tests PRIVATE boat_core)
add_test(NAME boat_wave_gyro_tests COMMAND boat_wave_gyro_tests)
add_executable(boat_sensor_tests tests/test_sensor_fixture.cpp)
target_link_libraries(boat_sensor_tests PRIVATE boat_core)
add_test(NAME boat_sensor_tests COMMAND boat_sensor_tests)
add_executable(boat_onboard_tests tests/test_onboard_navigation.cpp)
target_link_libraries(boat_onboard_tests PRIVATE boat_core)
add_test(NAME boat_onboard_tests COMMAND boat_onboard_tests)
add_executable(boat_shore_tests tests/test_shore_ranges.cpp)
target_link_libraries(boat_shore_tests PRIVATE boat_core)
add_test(NAME boat_shore_tests COMMAND boat_shore_tests)
add_executable(boat_association_tests tests/test_shore_association.cpp)
target_link_libraries(boat_association_tests PRIVATE boat_core)
add_test(NAME boat_association_tests COMMAND boat_association_tests)
add_executable(boat_marine_tests tests/test_marine_kinematics.cpp)
target_link_libraries(boat_marine_tests PRIVATE boat_core)
add_test(NAME boat_marine_tests COMMAND boat_marine_tests)
add_executable(boat_marine_dynamics_tests tests/test_marine_dynamics.cpp)
target_link_libraries(boat_marine_dynamics_tests PRIVATE boat_core)
add_test(NAME boat_marine_dynamics_tests COMMAND boat_marine_dynamics_tests)

if(BOAT_WITH_MAVLINK)
  if(UNIX)
    find_package(Threads REQUIRED)
    add_executable(boat_sitl_sensor apps/sitl_sensor_bridge.cpp)
    target_link_libraries(boat_sitl_sensor PRIVATE boat_core Threads::Threads)
    add_executable(boat_sitl_dynamic apps/sitl_dynamic_bridge.cpp)
    target_link_libraries(boat_sitl_dynamic PRIVATE boat_core Threads::Threads)
    add_executable(boat_sitl_bridge apps/sitl_bridge.cpp)
    target_link_libraries(boat_sitl_bridge PRIVATE boat_core Threads::Threads)

    add_executable(boat_sitl_link_tests tests/test_sitl_link.cpp)
    target_link_libraries(boat_sitl_link_tests PRIVATE boat_core Threads::Threads)
    target_include_directories(boat_sitl_link_tests SYSTEM PRIVATE "${BOAT_MAVLINK_INCLUDE_DIR}")
    add_test(NAME boat_sitl_link_tests COMMAND boat_sitl_link_tests)
  endif()

  add_executable(boat_mavlink_tests tests/test_mavlink.cpp)
  target_link_libraries(boat_mavlink_tests PRIVATE boat_core)
  target_include_directories(boat_mavlink_tests SYSTEM PRIVATE "${BOAT_MAVLINK_INCLUDE_DIR}")
  add_test(NAME boat_mavlink_tests COMMAND boat_mavlink_tests)
endif()
